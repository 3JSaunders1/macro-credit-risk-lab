# Makefile — common development commands
# Usage: make <target>

.PHONY: setup run api test docker-build docker-run docker-compose clean help

## ── Local Dev ──────────────────────────────────────────────────────────
setup:              ## Create venv and install dependencies
	python setup_env.py

run:                ## Run the full pipeline (CLI)
	python main.py

run-quiet:          ## Run pipeline without verbose output
	python main.py --quiet

api:                ## Start the FastAPI server (hot-reload)
	uvicorn api.server:app --reload --port 8000

api-prod:           ## Start API in production mode (no reload)
	uvicorn api.server:app --host 0.0.0.0 --port 8000

## ── Tests ──────────────────────────────────────────────────────────────
test:               ## Run all tests
	pytest tests/ -v

test-fast:          ## Run tests, stop on first failure
	pytest tests/ -x -v

test-cov:           ## Run tests with coverage report
	pytest tests/ --cov=. --cov-report=term-missing

## ── Docker ─────────────────────────────────────────────────────────────
docker-build:       ## Build the Docker image
	docker build -t credit-model .

docker-run:         ## Run the API in Docker
	docker run -p 8000:8000 credit-model

docker-run-cli:     ## Run the CLI pipeline in Docker
	docker run --rm credit-model python main.py

docker-compose:     ## Start services with docker-compose
	docker compose up --build

docker-compose-cli: ## Run CLI via docker-compose profile
	docker compose --profile cli up

## ── Utilities ──────────────────────────────────────────────────────────
structure:          ## Print project file structure
	python file_structure.py

clean:              ## Remove venv, pycache, pytest artifacts
	rm -rf venv __pycache__ .pytest_cache
	find . -type d -name __pycache__ -exec rm -rf {} + 2>/dev/null || true
	find . -name "*.pyc" -delete

help:               ## Show this help
	@grep -E '^[a-zA-Z_-]+:.*?## .*$$' $(MAKEFILE_LIST) | \
		awk 'BEGIN {FS = ":.*?## "}; {printf "  \033[36m%-20s\033[0m %s\n", $$1, $$2}'
